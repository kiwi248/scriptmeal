#LLM에 Tavily Tool Calling기능추가

# LLM에게 tool 정의를 알려줌
tools = [
    {
        "type": "function",
        "function": {
            "name": "search_recipe",
            "description": "모르는 음식이나 최신 유행 음식 레시피를 검색할 때 사용",
            "parameters": {
                "query": {"type": "string"}
            }
        }
    }
]

def get_recipe(user_query, chat_history):
    #유저쿼리임베딩 ex)두바이쫀득쿠키 다이어트 버전 알려줘
    query_embedding = openai.embed(user_query)
    
    #pgvector로 관련제품 5개만 뽑고 id, 브랜드, 제품명, 칼로리 정보만 갖고오기
    products = db.query("""
        SELECT p.id, p.brand, p.product_name, p.calories
        FROM product_embeddings pe
        JOIN products p ON pe.product_id = p.id
        ORDER BY pe.embedding <=> %s
        LIMIT 5
    """, query_embedding)

    system_prompt = f"""
        당신은 다이어트 레시피 챗봇입니다.
        아래 시판 제품 후보 중 적합한 1~2개를 골라 사용하세요.
        모르는 음식이거나 최신 유행 음식이면
        반드시 search_recipe tool을 먼저 호출하세요.
        
        응답은 반드시 JSON 형식으로:
        {{
            "recipe_text": "레시피 내용...",
            "used_product_ids": [1, 3]
        }}

        [사용 가능한 시판 제품]
        {products}
    """

    response = openai.chat.completions.create(
        model="gpt-4o-mini",
        tools=tools,  # tavily search_recipe tool
        messages=[
            {"role": "system", "content": system_prompt},
            *chat_history,
            {"role": "user", "content": user_query}
        ]
    )

    # LLM이 tool 호출을 결정했으면
    if response.choices[0].message.tool_calls:
        tool_call = response.choices[0].message.tool_calls[0]
        search_query = tool_call.function.arguments["query"]

        # Tavily로 실제 레시피 검색
        search_result = tavily.search(search_query)

        # 검색 결과 + 시판 제품 합쳐서 다시 LLM 호출
        response = openai.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": f"""
                    당신은 다이어트 레시피 챗봇입니다.
                    아래 검색된 레시피를 참고해서
                    시판 제품을 활용한 다이어트 버전으로 재구성하세요.
                    
                    [검색된 레시피 참고]
                    {search_result}
                    
                    [사용 가능한 시판 제품]
                    {products}
                """},
                *chat_history,
                {"role": "user", "content": user_query}
            ]
        )

    return parse_response(response)
