import os
import math
import requests
from datetime import datetime, timedelta
from openai import OpenAI

OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY", "")
KMA_API_KEY    = os.environ.get("KMA_API_KEY", "")

DEFAULT_LAT = 37.5665
DEFAULT_LON = 126.9780

# ── 위경도 → 기상청 격자 변환 ─────────────────────────────────────────────────

def latlon_to_grid(lat, lon):
    RE, GRID = 6371.00877, 5.0
    SLAT1, SLAT2, OLON, OLAT, XO, YO = 30.0, 60.0, 126.0, 38.0, 43, 136
    D = math.pi / 180.0
    re = RE / GRID
    sn = math.log(math.cos(SLAT1*D) / math.cos(SLAT2*D)) / \
         math.log(math.tan(math.pi*.25 + SLAT2*D*.5) / math.tan(math.pi*.25 + SLAT1*D*.5))
    sf = math.tan(math.pi*.25 + SLAT1*D*.5)**sn * math.cos(SLAT1*D) / sn
    ro = re * sf / math.tan(math.pi*.25 + OLAT*D*.5)**sn
    ra = re * sf / math.tan(math.pi*.25 + lat*D*.5)**sn
    theta = lon*D - OLON*D
    if theta >  math.pi: theta -= 2*math.pi
    if theta < -math.pi: theta += 2*math.pi
    return int(ra*math.sin(theta*sn) + XO + .5), int(ro - ra*math.cos(theta*sn) + YO + .5)

# ── 기상청 초단기실황 조회 ────────────────────────────────────────────────────

def fetch_weather(lat=DEFAULT_LAT, lon=DEFAULT_LON):
    now = datetime.now()
    if now.minute < 10:
        now -= timedelta(hours=1)
    nx, ny = latlon_to_grid(lat, lon)

    res = requests.get(
        "http://apis.data.go.kr/1360000/VilageFcstInfoService_2.0/getUltraSrtNcst",
        params={
            "serviceKey": KMA_API_KEY,
            "numOfRows": 10, "pageNo": 1, "dataType": "JSON",
            "base_date": now.strftime("%Y%m%d"),
            "base_time": now.strftime("%H00"),
            "nx": nx, "ny": ny,
        },
        timeout=10,
    ).json()

    obs = {i["category"]: i["obsrValue"]
           for i in res["response"]["body"]["items"]["item"]}

    temp    = float(obs.get("T1H", 0))
    pty     = obs.get("PTY", "0")
    weather = {"0":"맑음","1":"비","2":"비/눈","3":"눈","5":"빗방울","6":"빗방울/눈","7":"눈날림"}.get(pty, "맑음")
    return temp, weather

# ── 프롬프트 생성 ─────────────────────────────────────────────────────────────

def build_prompt(temp, weather):
    now = datetime.now()
    return f"""지금은 {now.strftime('%Y년 %m월 %d일 %H:%M')}이고 현재 날씨는 {weather}, 기온은 {temp}°C입니다.

위 상황을 반영해서 다이어트에 좋은 메뉴를 추천하고 간단한 레시피를 알려주세요.
- {now.month}월 대한민국 제철 재료 활용
- 날씨·기온에 맞게 (비/눈이면 따뜻한 국물, 30°C 이상이면 시원한 음식)
- 현재 시간에 적합한 양
- 칼로리 포함"""


# ── GPT 호출 ──────────────────────────────────────────────────────────────────

def ask(user_message, history=None):
    history = history or []
    temp, weather = fetch_weather()
    history.append({"role": "user", "content": user_message})

    res = OpenAI(api_key=OPENAI_API_KEY).chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "system", "content": build_prompt(temp, weather)}] + history,
        max_tokens=1000,
    )
    reply = res.choices[0].message.content
    history.append({"role": "assistant", "content": reply})
    return reply, history

# ── 실행 ──────────────────────────────────────────────────────────────────────

if __name__ == "__main__":
    print("=== 다이어트 레시피 챗봇 (exit 종료) ===\n")
    history = []
    while True:
        q = input("나: ").strip()
        if not q or q == "exit":
            break
        reply, history = ask(q, history)
        print(f"\n봇: {reply}\n")
